# To learn the theory behind finite state machines, regular expressions,
# grammar and parsers, I made a script language and a Parser project.
# This can be used to define new languages, tokens and LL(1) grammar,
# and generate the code for lexing and parsing this new language.

# This file will define a language inspired by Lean, but smaller with
# less syntactical sugar. Just a bare minimum to be able to verify
# proofs in mathematics. I want to understand how to make a theorem
# prover from scratch and write it in simple C++.

# Tokens that will be ignored
# ---------------------------

Ignore COMMENT = '#' [\01-\FF~\0A\0D]*;
Ignore SPACE   = [\09\20]+;
Ignore NL      = '\0A' | '\0D' | "\0D\0A" | "\0A\0D";

# Identifiers
# -----------

Token IDENT    = [_A-Za-z] [._0-9A-Za-z]*; # Lazy namespace by allowing .

# Keywords, never starts with lower case letter
# ---------------------------------------------

Token _Prop             = "Prop";

Token _Prop_True        = "Prop.True";
Token _Prop_False       = "Prop.False";

Token _RegisterEquality = "RegisterEquality";

Token _Def              = "Def";
Token _Let              = "Let";
Token _Match            = "Match";
Token _Case             = "Case";
Token _Else             = "Else";
Token _End              = "End";
Token _Expression       = "Expression";

Token _Theorem          = "Theorem";
Token _Proof            = "Proof";
Token _Eval             = "Eval";
Token _Trivial          = "Trivial";
Token _Induction        = "Induction";
Token _Rewrite          = "Rewrite";

Token _Inductive        = "Inductive";
Token _Type             = "Type";

# Symbols
# =======

Token COLONEQUAL    = ":=";
Token EQUAL         = '=';
Token OPEN_PAR      = '(';
Token CLOSE_PAR     = ')';
Token COLON         = ':';
Token SLIM_ARROW    = "->";
Token FAT_ARROW     = "=>";
Token PIPE          = '|';

# Grammar written in LL(1)
# ========================

# To avoid name collitions, I use ('<' alphanum* '>') for symbols, while
# tokens are (alphanum*). General tokens are UPPERCASE while fixed keywords
# are prefixed with _ followed by the keyword (if legal).
# That makes it easier to read as well.

# <Start> is the main entry point. An invisible temporary rule
# with a single production will call <Start> like this:
# Rule <> = <Start> END_OF_FILE;

Skip <Start>
    = <Item>;

Skip <Item>
    = <Inductive> <Item>
    | <Def> <Item>
    | <RegisterEquality> <Item>
    | <Theorem> <Item>
    | ;



Rule <Inductive>
    = _Inductive IDENT COLON _Type <ConstructorList> _End;

Skip <ConstructorList>
    = <Constructor> <ConstructorList>
    | ;

Rule <Constructor>
    = PIPE IDENT <ParamList>;



Rule <Def>
    = _Def IDENT <ParamList> COLON <TypeExpr> COLONEQUAL <Statements> _End;



# Dirty construction
Rule <RegisterEquality>
    = _RegisterEquality IDENT;



Rule <Theorem>
    = _Theorem IDENT <ParamList> COLON <ArgExpr> _Proof <Statements> _End;



# Some common rules that will be used in multiple places
# ---------------------------------------------------------------------



# Parameters for theorems and definitions, e.g. (p : Prop) (n : myNat)

Skip <ParamList>
    = <Param> <ParamList>
    | ;

Rule <Param>
    = OPEN_PAR IDENT COLON <TypeExpr> CLOSE_PAR;



Rule <TypeExpr>
    = _Type
    | _Prop
    | IDENT
    | OPEN_PAR <TypeExpr> SLIM_ARROW <TypeExpr> CLOSE_PAR;



# Arguments for functions and constructors, e.g. n1 or nat.succ(nat.add(n1 n3))

Rule <ArgExpr>
    = IDENT <ArgList>
    | _Eval <ArgList>
    | _Expression <ArgList>
    | _Prop_True
    | _Prop_False;

Skip <ArgList>
    = OPEN_PAR <Args> CLOSE_PAR
    | ;

Skip <Args>
    = <ArgExpr> <Args>
    | ;



#

Skip <Statements>
    = <Statement> <Statements>
    | ;

Rule <Statement>
    = _Let IDENT EQUAL <ArgExpr>
    | <ArgExpr>
    | _Match IDENT <Cases> _End
    | _Induction IDENT <ICases> _End
    | _Trivial
    | _Rewrite IDENT <ArgExpr> FAT_ARROW <ArgExpr>;

Skip <Cases>
    = <Case> <Cases>
    | _Else FAT_ARROW <Statements>
    | ;

Skip <Case>
    = _Case <ArgExpr> FAT_ARROW <Statements>;

Skip <ICases>
    = _Case <ArgExpr> <ICase> <ICases>
    | ;

Skip <ICase>
    = IDENT FAT_ARROW <Statements>
    | FAT_ARROW <Statements>;

# Proof checking model
# ====================
#
# The proof checker maintains:
#
# - A context of variables:
#     x : nat
#     p : Prop
#
# - A context of proof assumptions:
#     ...
#
# - A current goal:
#     goal : PropExpr
#
# Each ProofStep either:
# - modifies the current goal and context, or
# - adds a proof assumption, or
# - closes the current goal.
